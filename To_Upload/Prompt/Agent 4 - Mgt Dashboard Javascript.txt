// Read Agent-4 output.
const source=$input.first().json;
const payload=source.output??source;
const dashboard=payload.Dashboard??{};
const r4=Array.isArray(payload.R4)?payload.R4:[];

// Helpers.
const toNum=(v)=>{const n=Number(v);return Number.isFinite(n)?n:0;};
const esc=(v)=>String(v??'').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');
const normalizeRisk=(v)=>String(v??'').trim().toUpperCase();
const normalizeAction=(v)=>String(v??'').trim().toUpperCase().replace(/[\s-]+/g,'_');
const pct=(count,total)=>total>0?(count/total)*100:0;
const fmtPct=(v)=>`${Math.round(Number(v))}%`;
const riskRank={HIGH:3,MEDIUM:2,LOW:1};

// Calculate ALL portfolio metrics from the complete R4 array.
// LOW-risk customers remain part of these calculations.
const totalCustomers=r4.length;
const highRiskCount=r4.filter(r=>normalizeRisk(r.Overall_Risk_Level)==='HIGH').length;
const mediumRiskCount=r4.filter(r=>normalizeRisk(r.Overall_Risk_Level)==='MEDIUM').length;
const lowRiskCount=r4.filter(r=>normalizeRisk(r.Overall_Risk_Level)==='LOW').length;

const escalateCount=r4.filter(r=>normalizeAction(r.Recommended_Action)==='ESCALATE').length;
const manualReviewCount=r4.filter(r=>normalizeAction(r.Recommended_Action)==='MANUAL_REVIEW').length;
const approveCount=r4.filter(r=>normalizeAction(r.Recommended_Action)==='APPROVE').length;

const highRiskPct=pct(highRiskCount,totalCustomers);
const mediumRiskPct=pct(mediumRiskCount,totalCustomers);
const lowRiskPct=pct(lowRiskCount,totalCustomers);
const escalatePct=pct(escalateCount,totalCustomers);
const manualReviewPct=pct(manualReviewCount,totalCustomers);
const approvePct=pct(approveCount,totalCustomers);

// Qualitative Agent-4 content.
const executiveSummary=dashboard.Executive_Summary??'Portfolio risk assessment completed from consolidated specialist assessments.';
let redFlags=Array.isArray(dashboard.Top_Red_Flags)?dashboard.Top_Red_Flags.filter(Boolean):[];
let recommendations=Array.isArray(dashboard.Key_Recommendations)?dashboard.Key_Recommendations.filter(Boolean):[];

// Fallback red flags.
if(redFlags.length===0){
  const flags=[];
  for(const customer of r4){
    if(Array.isArray(customer.Main_Red_Flags))flags.push(...customer.Main_Red_Flags);
  }
  redFlags=[...new Set(flags.filter(Boolean))];
}

// Fallback recommendations.
if(recommendations.length===0){
  recommendations=[
    'Prioritize HIGH-risk customers for immediate escalation.',
    'Complete manual reviews for MEDIUM-risk customers.',
    'Maintain standard monitoring for LOW-risk customers.'
  ];
}

// Management table:
// Show ONLY HIGH and MEDIUM customers.
// R4 itself is NOT changed.
const priorityCustomers=[...r4]
  .filter(customer=>{
    const risk=normalizeRisk(customer.Overall_Risk_Level);
    return risk==='HIGH'||risk==='MEDIUM';
  })
  .sort((a,b)=>{
    const aRank=riskRank[normalizeRisk(a.Overall_Risk_Level)]??0;
    const bRank=riskRank[normalizeRisk(b.Overall_Risk_Level)]??0;
    if(bRank!==aRank)return bRank-aRank;
    return toNum(b.Overall_Risk_Score)-toNum(a.Overall_Risk_Score);
  })
  .slice(0,5);

// Text wrapping.
const wrapText=(text,maxCharacters=125,maxLines=3)=>{
  const words=String(text??'').split(/\s+/).filter(Boolean);
  const lines=[];
  let line='';
  for(const word of words){
    const candidate=line?`${line} ${word}`:word;
    if(candidate.length>maxCharacters&&line){
      lines.push(line);
      line=word;
      if(lines.length>=maxLines-1)break;
    }else{
      line=candidate;
    }
  }
  if(line&&lines.length<maxLines)lines.push(line);
  return lines;
};

// Donut chart.
const donut=(cx,cy,radius,percent,color,title,subtitle)=>{
  const safePercent=Math.max(0,Math.min(100,percent));
  const circumference=2*Math.PI*radius;
  const filled=circumference*safePercent/100;
  const remaining=circumference-filled;
  return `<text x='${cx}' y='${cy-radius-22}' text-anchor='middle' class='chart-title'>${esc(title)}</text><circle cx='${cx}' cy='${cy}' r='${radius}' fill='none' stroke='#E5E7EB' stroke-width='18'/><circle cx='${cx}' cy='${cy}' r='${radius}' fill='none' stroke='${color}' stroke-width='18' stroke-linecap='round' stroke-dasharray='${filled} ${remaining}' transform='rotate(-90 ${cx} ${cy})'/><text x='${cx}' y='${cy+11}' text-anchor='middle' class='donut-value'>${fmtPct(safePercent)}</text><text x='${cx}' y='${cy+radius+28}' text-anchor='middle' class='chart-subtitle'>${esc(subtitle)}</text>`;
};

// Bullet list.
const bulletList=(items,x,firstY,lineHeight,dotColor,maxItems=3,maxChars=68)=>items.slice(0,maxItems).map((item,index)=>{
  const y=firstY+index*lineHeight;
  return `<circle cx='${x}' cy='${y-4}' r='5' fill='${dotColor}'/><text x='${x+16}' y='${y}' class='bullet-text'>${esc(String(item).slice(0,maxChars))}</text>`;
}).join('');

const summarySvg=summaryLines.map((line,index)=>`<text x='60' y='${292+index*24}' class='summary-text'>${esc(line)}</text>`).join('');

// Customer rows.
// Column header ends at y=726.
// First customer row begins at y=728.
const tableRows=priorityCustomers.map((customer,index)=>{
  const rowTop=728+index*24;
  const textY=rowTop+16;
  const risk=normalizeRisk(customer.Overall_Risk_Level);
  const action=normalizeAction(customer.Recommended_Action);
  const riskColor=risk==='HIGH'?HIGH:MEDIUM;
  const actionColor=action==='ESCALATE'?HIGH:MEDIUM;
  const customerLabel=`${customer.Customer_ID??''}${customer.Customer_Name?` - ${customer.Customer_Name}`:''}`;
  return `<rect x='838' y='${rowTop}' width='704' height='24' fill='${index%2===0?'#FFFFFF':'#F7F9FC'}'/><line x1='838' y1='${rowTop+24}' x2='1542' y2='${rowTop+24}' stroke='#E4E7EB'/><text x='852' y='${textY}' class='table-row'>${esc(customerLabel.slice(0,34))}</text><text x='1110' y='${textY}' class='table-risk' fill='${riskColor}'>${esc(risk)}</text><text x='1245' y='${textY}' class='table-row'>${esc(customer.Overall_Risk_Score??'')}</text><text x='1350' y='${textY}' class='table-action' fill='${actionColor}'>${esc(String(customer.Recommended_Action??'').replace(/_/g,' '))}</text>`;
}).join('');

</style>

<rect x='0' y='0' width='1600' height='900' fill='${PAGE}'/>
